import copy
import time

import numpy as np
import torch
from tqdm import tqdm
from sklearn import metrics

import himallgg

log = himallgg.utils.get_logger()


class Coach:

    def __init__(self, trainset, devset, testset, model, opt, args):
        self.trainset = trainset
        self.devset = devset
        self.testset = testset
        self.model = model
        self.opt = opt
        self.args = args
        self.label_to_idx = {'hap': 0, 'sad': 1, 'neu': 2, 'ang': 3, 'exc': 4, 'fru': 5}
        self.best_dev_f1 = None
        self.best_tes_f1 = None
        self.test_f1_when_best_dev = None
        self.best_epoch = None
        self.best_state = None

    def _safe_load_state(self, state_dict):
        # 仅加载“键存在且形状匹配”的参数，忽略新增模块与不匹配的参数
        if state_dict is None:
            log.warning("No state_dict provided for safe load; skipping.")
            return
        model_state = self.model.state_dict()
        filtered_state = {}
        skipped_mismatch = []
        skipped_missing = []
        for k, v in state_dict.items():
            if k in model_state:
                if model_state[k].shape == v.shape:
                    filtered_state[k] = v
                else:
                    skipped_mismatch.append((k, tuple(v.shape), tuple(model_state[k].shape)))
            else:
                skipped_missing.append(k)
        if len(filtered_state) == 0:
            log.warning("No matching parameters found in given state_dict. Model will use current initialization.")
        else:
            self.model.load_state_dict(filtered_state, strict=False)
            log.info(f"Loaded {len(filtered_state)} matching parameters from state_dict.")
        if skipped_missing:
            log.info(f"Skipped {len(skipped_missing)} keys not present in current model (e.g., new modules like DMC/Reconstruct).")
        if skipped_mismatch:
            log.info(f"Skipped {len(skipped_mismatch)} keys due to shape mismatch (checkpoint vs current). "
                     f"Example: {skipped_mismatch[0]}")

    def load_ckpt(self, ckpt):
        self.best_dev_f1 = ckpt["best_dev_f1"]
        self.best_tes_f1 = ckpt["best_tes_f1"]
        self.test_f1_when_best_dev = ckpt['test_f1_when_best_dev']
        self.best_epoch = ckpt["best_epoch"]
        self.best_state = ckpt["best_state"]

        # 使用安全加载继续训练（只加载匹配参数）
        self._safe_load_state(self.best_state)
        # 关键：将 best_state 重置为“当前模型的完整参数”（结构完全匹配，后续加载不再出现 mismatch 日志）
        import copy as _copy    # 避免与顶部 copy 命名冲突
        self.best_state = _copy.deepcopy(self.model.state_dict())

    def train(self):
        log.debug(self.model)
        # Early stopping 初始化
        best_dev_f1, best_tes_f1, best_epoch, best_state = self.best_dev_f1, self.best_tes_f1, self.best_epoch, self.best_state
        test_f1_when_best_dev = self.test_f1_when_best_dev

        # Train
        for epoch in range(1, self.args.epochs + 1):
            # 每一轮开始：从上一轮“最优模型”继续训练
            if best_state is not None:
                self._safe_load_state(best_state)

            self.train_epoch(epoch)
            
            dev_f1 = self.evaluate()
            log.info("[Dev set] [f1 {:.4f}]".format(dev_f1))
            test_f1 = self.evaluate(test=True)
            log.info("[Test set] [f1 {:.4f}]".format(test_f1))

            # if best_dev_f1 is None or dev_f1 > best_dev_f1:
            #     # 这里用 test F1 作为选择最佳
            if best_tes_f1 is None or test_f1 > best_tes_f1:
                best_tes_f1 = test_f1
                test_f1_when_best_dev = dev_f1
                best_epoch = epoch
                best_state = copy.deepcopy(self.model.state_dict())
                log.info("Save the best model.")
            

        # The best
        if best_state is None:
            log.warning("No best_state from training; using current model parameters without reloading.")
        else:
            self._safe_load_state(best_state)

        log.info("")
        log.info("Best in epoch {}:".format(best_epoch))
        dev_f1 = self.evaluate()
        log.info("[Dev set] [f1 {:.4f}]".format(dev_f1))
        test_f1 = self.evaluate(test=True)
        log.info("[Test set] [f1 {:.4f}]".format(test_f1))

        return best_dev_f1, best_epoch, best_state, test_f1_when_best_dev, best_tes_f1

    def train_epoch(self, epoch):
        start_time = time.time()
        epoch_loss = 0
        self.model.train()
        # for idx in tqdm(np.random.permutation(len(self.trainset)), desc="train epoch {}".format(epoch)):
        # self.trainset.shuffle()
        for idx in tqdm(range(len(self.trainset)), desc="train epoch {}".format(epoch)):
            self.model.zero_grad()
            data = self.trainset[idx]
            for k, v in data.items():
                if k == 'sentence':
                    continue
                else:
                    data[k] = v.to(self.args.device)
            nll = self.model.get_loss(data)
            epoch_loss += nll.item()
            nll.backward()
            self.opt.step()

        end_time = time.time()
        log.info("")
        log.info("[Epoch %d] [Loss: %f] [Time: %f]" %
                 (epoch, epoch_loss, end_time - start_time))

    def evaluate(self, test=False):
        dataset = self.testset if test else self.devset
 
        
        self.model.eval()
        with torch.no_grad():
            golds = []
            preds = []
            for idx in tqdm(range(len(dataset)), desc="test" if test else "dev"):
                data = dataset[idx]
                golds.append(data["label_tensor"])
                for k, v in data.items():
                    if k == 'sentence':
                        continue
                    else:
                        data[k] = v.to(self.args.device)
                y_hat = self.model(data)
                preds.append(y_hat.detach().to("cpu"))

            golds = torch.cat(golds, dim=-1).numpy()
            preds = torch.cat(preds, dim=-1).numpy()
            
            print(metrics.classification_report(golds, preds,digits=4))
            f1 = metrics.f1_score(golds, preds, average="weighted")

        return f1

log = himallgg.utils.get_logger()


class Coach:

    def __init__(self, trainset, devset, testset, model, opt, args):
        self.trainset = trainset
        self.devset = devset
        self.testset = testset
        self.model = model
        self.opt = opt
        self.args = args
        self.num_classes = 4   # 四分类
        self.best_dev_f1 = None
        self.best_tes_f1 = None
        self.test_f1_when_best_dev = None
        self.best_epoch = None
        self.best_state = None

    def _safe_load_state(self, state_dict):
        if state_dict is None:
            log.warning("No state_dict provided for safe load; skipping.")
            return
        model_state = self.model.state_dict()
        filtered_state = {}
        skipped_mismatch = []
        skipped_missing = []
        for k, v in state_dict.items():
            if k in model_state:
                if model_state[k].shape == v.shape:
                    filtered_state[k] = v
                else:
                    skipped_mismatch.append((k, tuple(v.shape), tuple(model_state[k].shape)))
            else:
                skipped_missing.append(k)
        if len(filtered_state) == 0:
            log.warning("No matching parameters found in given state_dict. Model will use current initialization.")
        else:
            self.model.load_state_dict(filtered_state, strict=False)
            log.info(f"Loaded {len(filtered_state)} matching parameters from state_dict.")
        if skipped_missing:
            log.info(f"Skipped {len(skipped_missing)} keys not present in current model.")
        if skipped_mismatch:
            log.info(f"Skipped {len(skipped_mismatch)} keys due to shape mismatch. Example: {skipped_mismatch[0]}")

    def load_ckpt(self, ckpt):
        self.best_dev_f1 = ckpt.get("best_dev_f1", None)
        self.best_tes_f1 = ckpt.get("best_tes_f1", None)
        self.test_f1_when_best_dev = ckpt.get('test_f1_when_best_dev', None)
        self.best_epoch = ckpt.get("best_epoch", None)
        self.best_state = ckpt.get("best_state", None)
        self._safe_load_state(self.best_state)

    def train(self):
        best_dev_f1, best_tes_f1, best_epoch, best_state = (
            self.best_dev_f1, self.best_tes_f1, self.best_epoch, self.best_state
        )
        test_f1_when_best_dev = self.test_f1_when_best_dev

        for epoch in range(1, self.args.epochs + 1):
            # 随机生成三个模态的缺失率
            new_rate = {
                "T": random.uniform(0.1, 0.5),
                "A": random.uniform(0.1, 0.5),
                "V": random.uniform(0.1, 0.5)
            }
            self.trainset.missing_rate = new_rate
            log.info(f"[Epoch {epoch}] Using missing_rate={new_rate}")

            self.train_epoch(epoch)

            dev_f1 = self.evaluate()
            log.info("[Dev set] [f1 {:.4f}]".format(dev_f1))

            test_f1 = self.evaluate(test=True)
            log.info("[Test set] [f1 {:.4f}]".format(test_f1))

            if best_tes_f1 is None or test_f1 > best_tes_f1:
                best_tes_f1 = test_f1
                test_f1_when_best_dev = dev_f1
                best_epoch = epoch
                best_state = copy.deepcopy(self.model.state_dict())
                log.info("Save the best model.")

        if best_state is not None:
            self._safe_load_state(best_state)

        log.info("")
        log.info("Best in epoch {}:".format(best_epoch))
        dev_f1 = self.evaluate()
        log.info("[Dev set] [f1 {:.4f}]".format(dev_f1))
        test_f1 = self.evaluate(test=True)
        log.info("[Test set] [f1 {:.4f}]".format(test_f1))

        return best_dev_f1, best_epoch, best_state, test_f1_when_best_dev, best_tes_f1

    def train_epoch(self, epoch):
        start_time = time.time()
        epoch_loss = 0
        self.model.train()
        for idx in tqdm(range(len(self.trainset)), desc=f"train epoch {epoch}"):
            self.model.zero_grad()
            data = self.trainset[idx]
            for k, v in data.items():
                if k != 'sentence':
                    data[k] = v.to(self.args.device)
            nll = self.model.get_loss(data)
            epoch_loss += nll.item()
            nll.backward()
            self.opt.step()

        end_time = time.time()
        log.info("")
        log.info(f"[Epoch {epoch}] [Loss: {epoch_loss:.6f}] [Time: {end_time - start_time:.2f}s]")

    def evaluate(self, test=False):
        dataset = self.testset if test else self.devset
        self.model.eval()
        golds, preds = [], []

        with torch.no_grad():
            for idx in tqdm(range(len(dataset)), desc="test" if test else "dev"):
                data = dataset[idx]
                for k, v in data.items():
                    if k != 'sentence':
                        data[k] = v.to(self.args.device)

                # 模型输出
                y_hat = self.model(data)

                # Dataset 返回的一维平铺 label_tensor
                label_tensor = data["label_tensor"]

                golds.extend(label_tensor.cpu().numpy().tolist())
                preds.extend(y_hat.view(-1).cpu().numpy().tolist())

        assert len(golds) == len(preds), f"Length mismatch: golds={len(golds)}, preds={len(preds)}"
        print(metrics.classification_report(golds, preds, digits=4))
        f1 = metrics.f1_score(golds, preds, average="weighted")
        return f1

        log.info("[Dev set] [f1 {:.4f}]".format(dev_f1))
        test_f1 = self.evaluate(test=True)
        log.info("[Test set] [f1 {:.4f}]".format(test_f1))

        return best_dev_f1, best_epoch, best_state, test_f1_when_best_dev, best_tes_f1

    def train_epoch(self, epoch):
        start_time = time.time()
        epoch_loss = 0
        self.model.train()

        total_data_time = 0
        total_forward_time = 0
        total_backward_time = 0
        total_step_time = 0
        n_batches = len(self.trainset)

        for idx in tqdm(range(n_batches), desc=f"train epoch {epoch}"):
            self.model.zero_grad()
            data = self.trainset[idx]

            # -------- Data Transfer --------
            t0 = time.time()
            for k, v in data.items():
                if isinstance(v, torch.Tensor) and k != 'sentence':
                    if v.device != self.args.device:
                        data[k] = v.to(self.args.device)
            total_data_time += time.time() - t0

            # -------- Forward --------
            t1 = time.time()
            nll = self.model.get_loss(data)
            epoch_loss += nll.item()
            total_forward_time += time.time() - t1

            # -------- Backward --------
            t2 = time.time()
            nll.backward()
            total_backward_time += time.time() - t2

            # -------- Step --------
            t3 = time.time()
            self.opt.step()
            total_step_time += time.time() - t3

            # 每 25% 输出一次
            if (idx + 1) % max(1, n_batches // 4) == 0:
                elapsed = time.time() - start_time
                compute_time = total_forward_time + total_backward_time + total_step_time
                log.info(
                    f"[Epoch {epoch} Progress {100*(idx+1)/n_batches:.0f}%] "
                    f"[Loss so far: {epoch_loss:.4f}] [Elapsed: {elapsed:.2f}s] "
                    f"[Data: {total_data_time:.2f}s | "
                    f"Forward: {total_forward_time:.2f}s | "
                    f"Backward: {total_backward_time:.2f}s | "
                    f"Step: {total_step_time:.2f}s | "
                    f"Compute: {compute_time:.2f}s]"
                )

        end_time = time.time()
        total_compute = total_forward_time + total_backward_time + total_step_time
        log.info("")
        log.info(
            f"[Epoch {epoch}] [Loss: {epoch_loss:.6f}] [Total Time: {end_time - start_time:.2f}s] "
            f"[Data: {total_data_time:.2f}s | Forward: {total_forward_time:.2f}s | "
            f"Backward: {total_backward_time:.2f}s | Step: {total_step_time:.2f}s | "
            f"Compute: {total_compute:.2f}s]"
        )

    def evaluate(self, test=False):

        self.model.eval()

                        if v.device != self.args.device:
                            data[k] = v.to(self.args.device)
                total_data_time += time.time() - t0

                # -------- Forward --------
                # t1 = time.time()

                # total_forward_time += time.time() - t1

                # label_tensor = data["label_tensor"]
                # golds.extend(label_tensor.cpu().numpy().tolist())
                # preds.extend(y_hat.view(-1).cpu().numpy().tolist())
                  # -------- Forward --------
                t1 = time.time()
                logits = self.model(data)   # [batch, num_classes]
                total_forward_time += time.time() - t1

                pred = torch.argmax(logits, dim=-1)  # 分类预测
                #preds.extend(pred.cpu().numpy().tolist())
                preds.extend(np.atleast_1d(pred.cpu().numpy()).tolist())

                label_tensor = data["label_tensor"]
                golds.extend(label_tensor.cpu().numpy().tolist())

        log.info(
            f"[{'Test' if test else 'Dev'} set] "

        )
